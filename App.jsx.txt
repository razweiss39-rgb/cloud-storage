function formatBytes(bytes, decimals = 2) {
  if (!bytes || bytes === 0) return '0 B';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB', 'GB', 'TB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  const value = bytes / Math.pow(k, i);
  return `${value.toFixed(i === 0 ? 0 : decimals)} ${sizes[i]}`;
}

function getPlanById(planId) {
  return PLANS.find((p) => p.id === planId) || PLANS[0];
}

function buildLemonSqueezyUrl(baseUrl, user) {
  if (!baseUrl) return '#';
  try {
    const url = new URL(baseUrl);
    if (user?.email) url.searchParams.set('checkout[email]', user.email);
    if (user?.id) url.searchParams.set('checkout[custom][user_id]', user.id);
    return url.toString();
  } catch {
    return baseUrl;
  }
}

function generateApiKeyToken() {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  let token = 'sk_live_sv_';
  const array = new Uint32Array(32);
  window.crypto.getRandomValues(array);
  for (let i = 0; i < array.length; i++) {
    token += chars[array[i] % chars.length];
  }
  return token;
}

    setMobileMenuOpen(false);

          setUser(data?.session?.user || null);
          setAuthLoading(false);
        }
        const { data: listener } = supabase.auth.onAuthStateChange((_event, session) => {
          setUser(session?.user || null);
        });
        return () => listener?.subscription?.unsubscribe();
      } else {
        const saved = localStorage.getItem(DEMO_USER_KEY);
        if (mounted) {
          setUser(saved ? JSON.parse(saved) : null);
          setAuthLoading(false);
        }
      }
    }

    const cleanupPromise = initAuth();
    return () => {
      mounted = false;
      Promise.resolve(cleanupPromise).then((fn) => typeof fn === 'function' && fn());

      await supabase.auth.signOut();

      localStorage.removeItem(DEMO_USER_KEY);

        data: { ...user.user_metadata, plan_id: planId },

        setUser(data.user);
      }

      const updated = {
        ...user,
        user_metadata: { ...user.user_metadata, plan_id: planId },
      };

            {!HAS_WASABI && 'Wasabi S3 env vars not set (using local file simulation). '}
            Configure <code className="px-1 py-0.5 rounded bg-slate-900 text-indigo-300">VITE_*</code> variables in Cloudflare Pages for live production.

        {!['/', '/pricing', '/auth', '/dashboard', '/terms', '/privacy'].includes(route) && (

        )}

      {/* Footer */}
      <footer className="border-t border-slate-800/80 bg-slate-950 py-10 px-4 sm:px-6 lg:px-8">
        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-4 text-sm text-slate-400">
          <div className="flex items-center gap-2">
            <Cloud className="w-4 h-4 text-indigo-400" />

// AUTH PAGE (SUPABASE EMAIL/PASSWORD + MANDATORY TERMS CHECKBOX)
// ============================================================================
function AuthPage({ authMode, setAuthMode, user, setUser, navigate, showToast }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [acceptedTerms, setAcceptedTerms] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  useEffect(() => {
    if (user) navigate('/dashboard');
  }, [user, navigate]);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');

    if (!email.trim() || !password) {
      setError('Please enter both email and password.');
      return;
    }

    if (authMode === 'signup' && !acceptedTerms) {
      setError('You must accept the Terms of Service and Privacy Policy to create an account.');
      return;
    }

    setLoading(true);
    try {
      if (HAS_SUPABASE && supabase) {
        if (authMode === 'signup') {
          const { data, error: signUpError } = await supabase.auth.signUp({
            email: email.trim(),
            password,
            options: {
              data: {
                plan_id: 'free',
                accepted_terms_at: new Date().toISOString(),
              },
            },
          });
          if (signUpError) throw signUpError;

          if (data?.session?.user) {
            setUser(data.session.user);
            showToast('Welcome to SkyVault! Your 2 GB Free vault is ready.', 'success');
            navigate('/dashboard');
          } else {
            showToast('Check your email to confirm your account, then sign in.', 'success');
            setAuthMode('signin');
          }
        } else {
          const { data, error: signInError } = await supabase.auth.signInWithPassword({
            email: email.trim(),
            password,
          });
          if (signInError) throw signInError;
          setUser(data.user);
          showToast('Signed in successfully.', 'success');
          navigate('/dashboard');
        }
      } else {
        // Local Sandbox Auth
        const demoUser = {
          id: `demo_${btoa(email.trim()).replace(/[^a-zA-Z0-9]/g, '').slice(0, 12)}`,
          email: email.trim(),
          user_metadata: {
            plan_id: 'free',
            accepted_terms_at: new Date().toISOString(),
          },
        };
        localStorage.setItem(DEMO_USER_KEY, JSON.stringify(demoUser));
        setUser(demoUser);
        showToast(
          authMode === 'signup'
            ? 'Sandbox account created with 2 GB Free storage!'
            : 'Signed into Sandbox mode.',
          'success'
        );
        navigate('/dashboard');
      }
    } catch (err) {
      setError(err?.message || 'Authentication failed.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-md mx-auto py-16 px-4">
      <div className="rounded-2xl bg-slate-900/80 border border-slate-800 p-8 shadow-2xl">
        <div className="flex rounded-xl bg-slate-950 p-1 border border-slate-800 mb-6">
          <button
            type="button"
            onClick={() => {
              setAuthMode('signup');
              setError('');
            }}
            className={`flex-1 py-2 rounded-lg text-sm font-semibold transition cursor-pointer ${
              authMode === 'signup' ? 'bg-indigo-600 text-white' : 'text-slate-400 hover:text-white'
            }`}
          >
            Create Free Account
          </button>
          <button
            type="button"
            onClick={() => {
              setAuthMode('signin');
              setError('');
            }}
            className={`flex-1 py-2 rounded-lg text-sm font-semibold transition cursor-pointer ${
              authMode === 'signin' ? 'bg-indigo-600 text-white' : 'text-slate-400 hover:text-white'
            }`}
          >
            Sign In
          </button>
        </div>

        <h1 className="text-2xl font-bold text-white mb-1">
          {authMode === 'signup' ? 'Start Free with 2 GB' : 'Welcome back'}
        </h1>
        <p className="text-sm text-slate-400 mb-6">
          {authMode === 'signup'
            ? 'Create your encrypted SkyVault storage account.'
            : 'Sign in to manage your cloud files and transfers.'}
        </p>

        {error && (
          <div className="mb-4 p-3 rounded-xl bg-rose-950/60 border border-rose-500/40 text-rose-200 text-xs flex items-start gap-2">
            <AlertTriangle className="w-4 h-4 text-rose-400 shrink-0 mt-0.5" />
            <span>{error}</span>
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold uppercase tracking-wider text-slate-400 mb-1.5">
              Email Address
            </label>
            <input
              type="email"
              required
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              placeholder="you@example.com"
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:outline-none focus:border-indigo-500"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold uppercase tracking-wider text-slate-400 mb-1.5">
              Password
            </label>
            <input
              type="password"
              required
              minLength={6}
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              placeholder="Minimum 6 characters"
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:outline-none focus:border-indigo-500"
            />
          </div>

          {authMode === 'signup' && (
            <label className="flex items-start gap-3 p-3 rounded-xl bg-slate-950/70 border border-slate-800 cursor-pointer">
              <input
                type="checkbox"
                checked={acceptedTerms}
                onChange={(e) => setAcceptedTerms(e.target.checked)}
                required
                className="mt-1 h-4 w-4 rounded border-slate-700 bg-slate-900 text-indigo-600 focus:ring-indigo-500"
              />
              <span className="text-xs text-slate-300 leading-relaxed">
                I agree to the{' '}
                <button
                  type="button"
                  onClick={() => navigate('/terms')}
                  className="text-indigo-400 hover:underline font-medium cursor-pointer"
                >
                  Terms of Service
                </button>{' '}
                and{' '}
                <button
                  type="button"
                  onClick={() => navigate('/privacy')}
                  className="text-indigo-400 hover:underline font-medium cursor-pointer"
                >
                  Privacy Policy
                </button>
                , and confirm I will not upload prohibited executable files.
              </span>
            </label>
          )}

          <button
            type="submit"
            disabled={loading || (authMode === 'signup' && !acceptedTerms)}
            className="w-full py-3 rounded-xl bg-gradient-to-r from-indigo-600 to-cyan-600 hover:from-indigo-500 hover:to-cyan-500 disabled:opacity-50 disabled:cursor-not-allowed text-white font-bold text-sm transition cursor-pointer"
          >
            {loading
              ? 'Please wait...'
              : authMode === 'signup'
              ? 'Create Free 2 GB Account'
              : 'Sign In to Dashboard'}

        </form>
      </div>
    </div>
  );

// DASHBOARD PAGE (QUOTA BAR + WASABI S3 DRAG & DROP + SHARE LINKS + API KEYS)
// ============================================================================
function DashboardPage({ user, authLoading, currentPlan, navigate, showToast, onSimulatePlanChange }) {
  const [activeTab, setActiveTab] = useState('files');
  const [files, setFiles] = useState([]);
  const [loadingFiles, setLoadingFiles] = useState(false);
  const [uploading, setUploading] = useState(false);
  const [uploadStatus, setUploadStatus] = useState('');
  const [dragActive, setDragActive] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [shareModal, setShareModal] = useState(null);
  const [shareExpiry, setShareExpiry] = useState(86400); // 24 hours
  const [copiedId, setCopiedId] = useState(null);
  const [blockedAlert, setBlockedAlert] = useState([]);

  // API Keys state (Creator & Pro 1TB+ tiers)
  const [apiKeys, setApiKeys] = useState([]);
  const [newKeyName, setNewKeyName] = useState('');
  const [revealedKeyId, setRevealedKeyId] = useState(null);

  const fileInputRef = useRef(null);

  useEffect(() => {
    if (!authLoading && !user) {
      navigate('/auth', { authMode: 'signin' });
    }
  }, [authLoading, user, navigate]);

  const userPrefix = user ? `users/${user.id}/` : '';

  // Load API Keys from user metadata or local storage
  useEffect(() => {
    if (!user) return;
    if (HAS_SUPABASE && user.user_metadata?.api_keys) {
      setApiKeys(user.user_metadata.api_keys);
    } else {
      const saved = localStorage.getItem(`${DEMO_KEYS_KEY}_${user.id}`);
      setApiKeys(saved ? JSON.parse(saved) : []);
    }
  }, [user]);

  const persistApiKeys = async (nextKeys) => {
    setApiKeys(nextKeys);
    if (HAS_SUPABASE && supabase) {
      await supabase.auth.updateUser({
        data: { ...user.user_metadata, api_keys: nextKeys },
      });
    } else if (user) {
      localStorage.setItem(`${DEMO_KEYS_KEY}_${user.id}`, JSON.stringify(nextKeys));
    }
  };

  // Load Files from Wasabi S3 or Demo Store
  const fetchFiles = useCallback(async () => {
    if (!user) return;
    setLoadingFiles(true);
    try {
      if (HAS_WASABI && s3Client) {
        const res = await s3Client.send(
          new ListObjectsV2Command({
            Bucket: WASABI_BUCKET,
            Prefix: userPrefix,
          })
        );
        const items = (res.Contents || [])
          .filter((obj) => obj.Key && obj.Key !== userPrefix)
          .map((obj) => ({
            key: obj.Key,
            name: obj.Key.replace(userPrefix, ''),
            size: obj.Size || 0,
            lastModified: obj.LastModified ? new Date(obj.LastModified).toISOString() : new Date().toISOString(),
          }))
          .sort((a, b) => new Date(b.lastModified) - new Date(a.lastModified));
        setFiles(items);
      } else {
        const raw = localStorage.getItem(`${DEMO_FILES_KEY}_${user.id}`);
        setFiles(raw ? JSON.parse(raw) : []);
      }
    } catch (err) {
      showToast(`Failed to list files: ${err.message}`, 'error');
    } finally {
      setLoadingFiles(false);
    }
  }, [user, userPrefix, showToast]);

  useEffect(() => {
    fetchFiles();
  }, [fetchFiles]);

  const totalUsedBytes = useMemo(
    () => files.reduce((acc, file) => acc + (Number(file.size) || 0), 0),
    [files]
  );

  const quotaPct = useMemo(() => {
    if (!currentPlan.quotaBytes) return 0;
    return Math.min(100, (totalUsedBytes / currentPlan.quotaBytes) * 100);
  }, [totalUsedBytes, currentPlan.quotaBytes]);

  const filteredFiles = useMemo(() => {
    if (!searchQuery.trim()) return files;
    const q = searchQuery.toLowerCase();
    return files.filter((f) => f.name.toLowerCase().includes(q));
  }, [files, searchQuery]);

  // Upload Files Handler with Extension Blocking & Quota Enforcement
  const processUpload = async (fileList) => {
    if (!fileList || fileList.length === 0 || !user) return;
    const incoming = Array.from(fileList);

    const rejected = [];
    const allowed = [];

    for (const file of incoming) {
      if (BLOCKED_REGEX.test(file.name)) {
        rejected.push(file.name);
      } else {
        allowed.push(file);
      }
    }

    if (rejected.length > 0) {
      setBlockedAlert(rejected);
      showToast(
        `Blocked executable file(s): ${rejected.join(', ')}. Allowed storage excludes ${BLOCKED_EXTENSIONS.join(', ')}.`,
        'error'
      );
    } else {
      setBlockedAlert([]);
    }

    if (allowed.length === 0) return;

    const batchSize = allowed.reduce((sum, f) => sum + f.size, 0);
    if (totalUsedBytes + batchSize > currentPlan.quotaBytes) {
      showToast(
        `Upload exceeds your ${currentPlan.name} (${currentPlan.storageGB} GB) storage limit. Please upgrade your plan.`,
        'error'
      );
      return;
    }

    setUploading(true);
    try {
      for (let i = 0; i < allowed.length; i++) {
        const file = allowed[i];
        setUploadStatus(`Uploading ${file.name} (${i + 1}/${allowed.length})...`);
        const objectKey = `${userPrefix}${file.name}`;

        if (HAS_WASABI && s3Client) {
          const buffer = new Uint8Array(await file.arrayBuffer());
          await s3Client.send(
            new PutObjectCommand({
              Bucket: WASABI_BUCKET,
              Key: objectKey,
              Body: buffer,
              ContentType: file.type || 'application/octet-stream',
            })
          );
        } else {
          // Local Demo simulation
          const existingRaw = localStorage.getItem(`${DEMO_FILES_KEY}_${user.id}`);
          const existing = existingRaw ? JSON.parse(existingRaw) : [];
          const next = [
            {
              key: objectKey,
              name: file.name,
              size: file.size,
              type: file.type || 'application/octet-stream',
              lastModified: new Date().toISOString(),
            },
            ...existing.filter((f) => f.key !== objectKey),
          ];
          localStorage.setItem(`${DEMO_FILES_KEY}_${user.id}`, JSON.stringify(next));
        }
      }

      showToast(
        `Successfully uploaded ${allowed.length} file${allowed.length > 1 ? 's' : ''}.`,
        'success'
      );
      await fetchFiles();
    } catch (err) {
      showToast(`Upload error: ${err.message}`, 'error');
    } finally {
      setUploading(false);
      setUploadStatus('');
      if (fileInputRef.current) fileInputRef.current.value = '';
    }
  };

  const handleDeleteFile = async (file) => {
    try {
      if (HAS_WASABI && s3Client) {
        await s3Client.send(
          new DeleteObjectCommand({
            Bucket: WASABI_BUCKET,
            Key: file.key,
          })
        );
      } else {
        const next = files.filter((f) => f.key !== file.key);
        localStorage.setItem(`${DEMO_FILES_KEY}_${user.id}`, JSON.stringify(next));
      }
      setFiles((prev) => prev.filter((f) => f.key !== file.key));
      showToast(`Deleted "${file.name}".`, 'info');
    } catch (err) {
      showToast(`Failed to delete file: ${err.message}`, 'error');
    }
  };

  const handleGenerateShareLink = async (file, expirySeconds = shareExpiry) => {
    try {
      let url = '';
      if (HAS_WASABI && s3Client) {
        const command = new GetObjectCommand({
          Bucket: WASABI_BUCKET,
          Key: file.key,
          ResponseContentDisposition: `attachment; filename="${encodeURIComponent(file.name)}"`,
        });
        url = await getSignedUrl(s3Client, command, { expiresIn: expirySeconds });
      } else {
        const token = btoa(`${file.key}:${Date.now()}`).slice(0, 24);
        url = `${window.location.origin}/share/${token}?file=${encodeURIComponent(file.name)}&exp=${expirySeconds}`;
      }
      setShareModal({ file, url, expirySeconds });
    } catch (err) {
      showToast(`Failed to generate link: ${err.message}`, 'error');
    }
  };

  const copyToClipboard = async (text, id = 'link') => {
    try {
      await navigator.clipboard.writeText(text);
      setCopiedId(id);
      showToast('Copied to clipboard!', 'success');
      setTimeout(() => setCopiedId(null), 2000);
    } catch {
      showToast('Could not access clipboard. Copy manually.', 'error');
    }
  };

  const handleCreateApiKey = async (e) => {
    e.preventDefault();
    if (!newKeyName.trim()) return;
    const newEntry = {
      id: `key_${Date.now()}`,
      name: newKeyName.trim(),
      token: generateApiKeyToken(),
      createdAt: new Date().toISOString(),
    };
    await persistApiKeys([newEntry, ...apiKeys]);
    setNewKeyName('');
    setRevealedKeyId(newEntry.id);
    showToast('API Key generated!', 'success');
  };

  const handleRevokeApiKey = async (id) => {
    await persistApiKeys(apiKeys.filter((k) => k.id !== id));
    showToast('API Key revoked.', 'info');
  };

  if (authLoading || !user) {
    return (
      <div className="py-24 text-center text-slate-400">Loading your cloud vault...</div>
    );
  }

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8 space-y-6">
      {/* Top User & Quota Card */}
      <div className="rounded-2xl bg-slate-900/70 border border-slate-800 p-6">
        <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-6">
          <div>
            <div className="flex items-center gap-2.5">
              <h1 className="text-2xl font-bold text-white">Cloud Storage Vault</h1>
              <span className="px-2.5 py-0.5 rounded-full text-xs font-bold bg-indigo-500/20 text-indigo-300 border border-indigo-500/30">
                {currentPlan.name} Plan
              </span>
            </div>
            <p className="text-sm text-slate-400 mt-1">
              Signed in as <span className="text-slate-200 font-medium">{user.email}</span> • Endpoint:{' '}
              <code className="text-cyan-300 text-xs">{WASABI_ENDPOINT}</code>
            </p>
          </div>

          {/* Storage Quota Progress Bar */}
          <div className="w-full lg:w-96 bg-slate-950/80 border border-slate-800 rounded-xl p-4">
            <div className="flex items-center justify-between text-xs font-semibold mb-2">
              <span className="text-slate-300 flex items-center gap-1.5">
                <HardDrive className="w-4 h-4 text-indigo-400" />
                Storage Quota
              </span>
              <span className="text-white">
                {formatBytes(totalUsedBytes)} / {currentPlan.storageGB} GB
              </span>
            </div>
            <div className="w-full h-2.5 rounded-full bg-slate-800 overflow-hidden">
              <div
                className={`h-full rounded-full transition-all duration-500 ${
                  quotaPct > 90
                    ? 'bg-rose-500'
                    : quotaPct > 75
                    ? 'bg-amber-500'
                    : 'bg-gradient-to-r from-indigo-500 to-cyan-400'
                }`}
                style={{ width: `${Math.max(quotaPct, totalUsedBytes > 0 ? 2 : 0)}%` }}
              />
            </div>
            <div className="mt-2 flex items-center justify-between text-[11px] text-slate-400">
              <span>{quotaPct.toFixed(1)}% used</span>
              <button
                onClick={() => navigate('/pricing')}
                className="text-indigo-400 hover:text-indigo-300 font-semibold cursor-pointer"
              >
                Upgrade Storage →
              </button>
            </div>
          </div>
        </div>

        {/* Dashboard Tabs */}
        <div className="mt-6 pt-4 border-t border-slate-800 flex flex-wrap gap-2">
          <button
            onClick={() => setActiveTab('files')}
            className={`px-4 py-2 rounded-lg text-sm font-semibold flex items-center gap-2 transition cursor-pointer ${
              activeTab === 'files'
                ? 'bg-indigo-600 text-white'
                : 'text-slate-400 hover:text-white hover:bg-slate-800'
            }`}
          >
            <FolderOpen className="w-4 h-4" />
            Files &amp; Transfers
          </button>

          <button
            onClick={() => setActiveTab('apikeys')}
            className={`px-4 py-2 rounded-lg text-sm font-semibold flex items-center gap-2 transition cursor-pointer ${
              activeTab === 'apikeys'
                ? 'bg-indigo-600 text-white'
                : 'text-slate-400 hover:text-white hover:bg-slate-800'
            }`}
          >
            <Key className="w-4 h-4" />
            API Keys
            {!currentPlan.hasApiKeys && <Lock className="w-3.5 h-3.5 text-amber-400" />}
          </button>

          <button
            onClick={() => navigate('/pricing')}
            className="px-4 py-2 rounded-lg text-sm font-semibold text-slate-400 hover:text-white hover:bg-slate-800 flex items-center gap-2 transition cursor-pointer"
          >
            <CreditCard className="w-4 h-4" />
            Plans &amp; Billing
          </button>
        </div>
      </div>

      {/* TAB 1: FILES & DRAG-AND-DROP UPLOAD */}
      {activeTab === 'files' && (
        <div className="space-y-6">
          {/* Blocked Files Warning */}
          {blockedAlert.length > 0 && (
            <div className="p-4 rounded-xl bg-rose-950/60 border border-rose-500/40 flex items-start justify-between gap-3">
              <div className="flex items-start gap-3">
                <ShieldAlert className="w-5 h-5 text-rose-400 shrink-0 mt-0.5" />
                <div className="text-sm text-rose-100">
                  <p className="font-bold">Security Policy: Executable Upload Blocked</p>
                  <p className="text-xs text-rose-200/80 mt-0.5">
                    Rejected: <strong>{blockedAlert.join(', ')}</strong>. Files ending in{' '}
                    <code>{BLOCKED_EXTENSIONS.join(', ')}</code> are strictly prohibited.
                  </p>
                </div>
              </div>
              <button onClick={() => setBlockedAlert([])} className="text-rose-300 hover:text-white">
                <X className="w-4 h-4" />
              </button>
            </div>
          )}

          {/* Drag and Drop Zone */}
          <div
            onDragOver={(e) => {
              e.preventDefault();
              setDragActive(true);
            }}
            onDragLeave={() => setDragActive(false)}
            onDrop={(e) => {
              e.preventDefault();
              setDragActive(false);
              processUpload(e.dataTransfer.files);
            }}
            onClick={() => !uploading && fileInputRef.current?.click()}
            className={`rounded-2xl border-2 border-dashed p-10 text-center transition cursor-pointer ${
              dragActive
                ? 'border-cyan-400 bg-cyan-950/20'
                : 'border-slate-700 bg-slate-900/40 hover:border-indigo-500/60 hover:bg-slate-900/70'
            }`}
          >
            <input
              ref={fileInputRef}
              type="file"
              multiple
              className="hidden"
              onChange={(e) => processUpload(e.target.files)}
            />
            <div className="w-14 h-14 mx-auto rounded-2xl bg-indigo-500/15 border border-indigo-500/30 flex items-center justify-center mb-4">
              <Upload className={`w-7 h-7 text-indigo-400 ${uploading ? 'animate-bounce' : ''}`} />
            </div>
            <h2 className="text-lg font-bold text-white">
              {uploading ? uploadStatus : 'Drag & drop files here, or click to browse'}
            </h2>
            <p className="text-sm text-slate-400 mt-1">
              Uploads via <code className="text-slate-300">@aws-sdk/client-s3</code> to Wasabi. Blocked:{' '}
              <span className="text-rose-300">{BLOCKED_EXTENSIONS.join(', ')}</span>
            </p>
          </div>

          {/* File List Table */}
          <div className="rounded-2xl bg-slate-900/70 border border-slate-800 overflow-hidden">
            <div className="p-4 border-b border-slate-800 flex flex-col sm:flex-row sm:items-center justify-between gap-3">
              <div className="flex items-center gap-2">
                <h3 className="font-bold text-white">Your Stored Files ({files.length})</h3>
                <button
                  onClick={fetchFiles}
                  className="p-1.5 rounded-lg text-slate-400 hover:text-white hover:bg-slate-800 cursor-pointer"
                  title="Refresh files"
                >
                  <RefreshCw className={`w-4 h-4 ${loadingFiles ? 'animate-spin' : ''}`} />
                </button>
              </div>
              <div className="relative w-full sm:w-64">
                <Search className="w-4 h-4 text-slate-400 absolute left-3 top-1/2 -translate-y-1/2" />
                <input
                  type="text"
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  placeholder="Search files..."
                  className="w-full pl-9 pr-3 py-1.5 rounded-lg bg-slate-950 border border-slate-800 text-sm text-white focus:outline-none focus:border-indigo-500"
                />
              </div>
            </div>

            {filteredFiles.length === 0 ? (
              <div className="p-12 text-center text-slate-400 text-sm">
                {files.length === 0
                  ? 'Your vault is empty. Drop files above to start uploading.'
                  : 'No files match your search.'}
              </div>
            ) : (
              <div className="overflow-x-auto">
                <table className="w-full text-left text-sm">
                  <thead className="bg-slate-950/60 text-xs uppercase text-slate-400 border-b border-slate-800">
                    <tr>
                      <th className="px-5 py-3">File Name</th>
                      <th className="px-5 py-3">Size</th>
                      <th className="px-5 py-3">Uploaded</th>
                      <th className="px-5 py-3 text-right">Actions</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-800/70">
                    {filteredFiles.map((file) => (
                      <tr key={file.key} className="hover:bg-slate-800/30 transition">
                        <td className="px-5 py-3.5 font-medium text-white flex items-center gap-2.5">
                          <FileIcon className="w-4 h-4 text-indigo-400 shrink-0" />
                          <span className="truncate max-w-xs sm:max-w-md">{file.name}</span>
                        </td>
                        <td className="px-5 py-3.5 text-slate-300">{formatBytes(file.size)}</td>
                        <td className="px-5 py-3.5 text-slate-400 text-xs">
                          {new Date(file.lastModified).toLocaleDateString()}
                        </td>
                        <td className="px-5 py-3.5 text-right">
                          <div className="inline-flex items-center gap-2">
                            <button
                              onClick={() => handleGenerateShareLink(file)}
                              className="px-3 py-1.5 rounded-lg bg-indigo-500/15 hover:bg-indigo-500/25 text-indigo-300 text-xs font-semibold inline-flex items-center gap-1.5 border border-indigo-500/25 cursor-pointer"
                            >
                              <Share2 className="w-3.5 h-3.5" />
                              Share Link
                            </button>
                            <button
                              onClick={() => handleDeleteFile(file)}
                              className="p-1.5 rounded-lg text-slate-400 hover:text-rose-400 hover:bg-rose-950/40 cursor-pointer"
                              title="Delete file"
                            >
                              <Trash2 className="w-4 h-4" />
                            </button>
                          </div>
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            )}
          </div>
        </div>
      )}

      {/* TAB 2: API KEYS (CREATOR 500GB & PRO 1TB+ TIERS) */}
      {activeTab === 'apikeys' && (
        <div className="rounded-2xl bg-slate-900/70 border border-slate-800 p-6">
          {!currentPlan.hasApiKeys ? (
            <div className="max-w-lg mx-auto py-10 text-center">
              <div className="w-14 h-14 mx-auto rounded-2xl bg-amber-500/15 border border-amber-500/30 flex items-center justify-center mb-4">
                <Lock className="w-7 h-7 text-amber-400" />
              </div>
              <h2 className="text-xl font-bold text-white">API Keys Require Creator or Pro Plans</h2>
              <p className="text-sm text-slate-400 mt-2 mb-6">
                Programmatic uploads, signed URL generation, and API key management are unlocked on{' '}
                <strong className="text-white">Creator (500 GB)</strong>,{' '}
                <strong className="text-white">Pro 1TB</strong>,{' '}
                <strong className="text-white">Pro Plus</strong>, and{' '}
                <strong className="text-white">Enterprise</strong>.
              </p>
              <div className="flex flex-wrap justify-center gap-3">
                <button
                  onClick={() => navigate('/pricing')}
                  className="px-6 py-2.5 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white font-semibold text-sm cursor-pointer"
                >
                  Upgrade to Creator ($6.99/mo)
                </button>
                {!HAS_SUPABASE && (
                  <button
                    onClick={() => onSimulatePlanChange('creator')}
                    className="px-4 py-2.5 rounded-xl bg-slate-800 hover:bg-slate-700 text-slate-200 text-sm font-medium cursor-pointer"
                  >
                    Preview Creator Tier (Demo)
                  </button>
                )}
              </div>
            </div>
          ) : (
            <div className="space-y-6">
              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
                <div>
                  <h2 className="text-xl font-bold text-white flex items-center gap-2">
                    <Key className="w-5 h-5 text-cyan-400" />
                    Developer API Keys
                  </h2>
                  <p className="text-sm text-slate-400 mt-1">
                    Use API keys to authenticate automated file transfers with your {currentPlan.name} vault.
                  </p>
                </div>
              </div>

              <form onSubmit={handleCreateApiKey} className="flex flex-col sm:flex-row gap-3">
                <input
                  type="text"
                  value={newKeyName}
                  onChange={(e) => setNewKeyName(e.target.value)}
                  placeholder="Key label (e.g., Production CI Backup)"
                  className="flex-1 px-4 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-sm text-white focus:outline-none focus:border-indigo-500"
                />
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white font-semibold text-sm flex items-center justify-center gap-2 cursor-pointer"
                >
                  <Plus className="w-4 h-4" />
                  Generate API Key
                </button>
              </form>

              {apiKeys.length === 0 ? (
                <div className="p-8 rounded-xl bg-slate-950/60 border border-slate-800 text-center text-sm text-slate-400">
                  No API keys created yet. Generate your first token above.
                </div>
              ) : (
                <div className="space-y-3">
                  {apiKeys.map((item) => {
                    const isRevealed = revealedKeyId === item.id;
                    const masked = `${item.token.slice(0, 14)}••••••••••••••••${item.token.slice(-4)}`;
                    return (
                      <div
                        key={item.id}
                        className="p-4 rounded-xl bg-slate-950/80 border border-slate-800 flex flex-col sm:flex-row sm:items-center justify-between gap-4"
                      >
                        <div>
                          <div className="font-semibold text-white text-sm">{item.name}</div>
                          <div className="font-mono text-xs text-cyan-300 mt-1 break-all">
                            {isRevealed ? item.token : masked}
                          </div>
                          <div className="text-[11px] text-slate-500 mt-1">
                            Created {new Date(item.createdAt).toLocaleString()}
                          </div>
                        </div>
                        <div className="flex items-center gap-2">
                          <button
                            onClick={() => setRevealedKeyId(isRevealed ? null : item.id)}
                            className="p-2 rounded-lg bg-slate-900 hover:bg-slate-800 text-slate-300 cursor-pointer"
                            title={isRevealed ? 'Hide key' : 'Reveal key'}
                          >
                            {isRevealed ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
                          </button>
                          <button
                            onClick={() => copyToClipboard(item.token, item.id)}
                            className="px-3 py-2 rounded-lg bg-slate-900 hover:bg-slate-800 text-slate-200 text-xs font-medium inline-flex items-center gap-1.5 cursor-pointer"
                          >
                            {copiedId === item.id ? (
                              <Check className="w-3.5 h-3.5 text-emerald-400" />
                            ) : (
                              <Copy className="w-3.5 h-3.5" />
                            )}
                            Copy
                          </button>
                          <button
                            onClick={() => handleRevokeApiKey(item.id)}
                            className="px-3 py-2 rounded-lg bg-rose-950/40 hover:bg-rose-950/70 text-rose-300 text-xs font-medium cursor-pointer"
                          >
                            Revoke
                          </button>
                        </div>
                      </div>
                    );
                  })}
                </div>
              )}
            </div>
          )}
        </div>
      )}

      {/* Shareable Download Link Modal */}
      {shareModal && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="w-full max-w-lg rounded-2xl bg-slate-900 border border-slate-700 p-6 shadow-2xl">
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-lg font-bold text-white flex items-center gap-2">
                <Share2 className="w-5 h-5 text-cyan-400" />
                Shareable Download Link
              </h3>
              <button onClick={() => setShareModal(null)} className="text-slate-400 hover:text-white">
                <X className="w-5 h-5" />
              </button>
            </div>

            <p className="text-sm text-slate-300 mb-4">
              File: <strong className="text-white">{shareModal.file.name}</strong> (
              {formatBytes(shareModal.file.size)})
            </p>

            <div className="mb-4">
              <label className="block text-xs font-semibold text-slate-400 uppercase mb-1.5 flex items-center gap-1.5">
                <Clock className="w-3.5 h-3.5" /> Link Expiration
              </label>
              <select
                value={shareExpiry}
                onChange={(e) => {
                  const val = Number(e.target.value);
                  setShareExpiry(val);
                  handleGenerateShareLink(shareModal.file, val);
                }}
                className="w-full px-3 py-2 rounded-lg bg-slate-950 border border-slate-800 text-sm text-white"
              >
                <option value={3600}>1 Hour</option>
                <option value={86400}>24 Hours</option>
                <option value={604800}>7 Days</option>
              </select>
            </div>

            <div className="p-3 rounded-xl bg-slate-950 border border-slate-800 font-mono text-xs text-cyan-300 break-all max-h-28 overflow-y-auto mb-4">
              {shareModal.url}
            </div>

            <div className="flex justify-end gap-2">
              {HAS_WASABI && (
                <a
                  href={shareModal.url}
                  target="_blank"
                  rel="noopener noreferrer"
                  className="px-4 py-2 rounded-xl bg-slate-800 hover:bg-slate-700 text-white text-sm font-semibold inline-flex items-center gap-1.5"
                >
                  <Download className="w-4 h-4" />
                  Test Download
                </a>
              )}
              <button
                onClick={() => copyToClipboard(shareModal.url, 'modal')}
                className="px-5 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-sm font-semibold inline-flex items-center gap-1.5 cursor-pointer"
              >
                {copiedId === 'modal' ? <Check className="w-4 h-4" /> : <Copy className="w-4 h-4" />}
                Copy Link
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
